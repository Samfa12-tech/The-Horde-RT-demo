COMPONENTS = {5121:('B',1),5123:('H',2),5125:('I',4),5126:('f',4)}

def sha(data): return hashlib.sha256(data).hexdigest()
def save_json(name, value):

class Glb:
    def __init__(self, path):
        self.raw = Path(path).read_bytes()
        magic, version, length = struct.unpack_from('<III', self.raw)
        assert magic == 0x46546c67 and version == 2 and length == len(self.raw)
        jlen, kind = struct.unpack_from('<II', self.raw, 12)
        assert kind == 0x4e4f534a
        self.doc = json.loads(self.raw[20:20+jlen])
        blen, kind = struct.unpack_from('<II', self.raw, 20+jlen)
        assert kind == 0x004e4942
        self.bin = self.raw[28+jlen:28+jlen+blen]
        assert len(self.bin) == blen and len(self.doc['buffers']) == 1

    def records(self, accessor):
        a = self.doc['accessors'][accessor]
        assert not a.get('sparse')
        v = self.doc['bufferViews'][a['bufferView']]
        width = COMPONENTS[a['componentType']][1] * WIDTH[a['type']]
        stride = v.get('byteStride', width)
        start = v.get('byteOffset',0) + a.get('byteOffset',0)
        result = [self.bin[start+i*stride:start+i*stride+width] for i in range(a['count'])]
        assert all(len(record) == width for record in result)
        return result

    def values(self, accessor):
        a = self.doc['accessors'][accessor]
        fmt = '<' + COMPONENTS[a['componentType']][0] * WIDTH[a['type']]
        return [struct.unpack(fmt, item) for item in self.records(accessor)]


def write_glb(path, doc, binary):
    encoded = json.dumps(doc,separators=(',',':'),ensure_ascii=False).encode()
    encoded += b' ' * (-len(encoded) % 4)
    binary = bytes(binary) + b'\0' * (-len(binary) % 4)
    raw = struct.pack('<III',0x46546c67,2,28+len(encoded)+len(binary))
    raw += struct.pack('<II',len(encoded),0x4e4f534a)+encoded
    raw += struct.pack('<II',len(binary),0x004e4942)+binary
    with path.open('xb') as stream: stream.write(raw)


def append_accessor(doc, binary, source_accessor, records, values=None):
    binary.extend(b'\0' * (-len(binary) % 4))
    offset = len(binary)
    binary.extend(b''.join(records))
    view = len(doc['bufferViews'])
    doc['bufferViews'].append({'buffer':0,'byteOffset':offset,'byteLength':len(binary)-offset})
    source = copy.deepcopy(source_accessor)
    source.update(bufferView=view, byteOffset=0, count=len(records))
    source.pop('min',None);source.pop('max',None)
    if values is not None and values:
        source['min'] = [min(v[i] for v in values) for i in range(len(values[0]))]
        source['max'] = [max(v[i] for v in values) for i in range(len(values[0]))]
    index = len(doc['accessors']);doc['accessors'].append(source)
    return index


def pkey(p): return tuple(round(x/1e-6) for x in p)
def edge(a,b): return tuple(sorted((a,b)))
def tri_edges(tri):
    return (edge(tri[0],tri[1]),edge(tri[1],tri[2]),edge(tri[2],tri[0]))
def triangles(indices): return [tuple(indices[i:i+3]) for i in range(0,len(indices),3)]
def boundary(positions, tris):
    counts = collections.Counter(e for t in tris for e in tri_edges(tuple(pkey(positions[i]) for i in t)))
    assert not any(a==b for a,b in counts)
    return collections.Counter({e:n for e,n in counts.items() if n==1})


def local_matrix(node):
    if 'matrix' in node: return node['matrix']
    x,y,z,w = node.get('rotation',[0,0,0,1]);sx,sy,sz=node.get('scale',[1,1,1]);t=node.get('translation',[0,0,0])
    return [(1-2*y*y-2*z*z)*sx,(2*x*y+2*w*z)*sx,(2*x*z-2*w*y)*sx,0,
            (2*x*y-2*w*z)*sy,(1-2*x*x-2*z*z)*sy,(2*y*z+2*w*x)*sy,0,
            (2*x*z+2*w*y)*sz,(2*y*z-2*w*x)*sz,(1-2*x*x-2*y*y)*sz,0,*t,1]
def mul(a,b): return [sum(a[k*4+row]*b[col*4+k] for k in range(4)) for col in range(4) for row in range(4)]
def node_worlds(doc):
    parents={child:i for i,n in enumerate(doc['nodes']) for child in n.get('children',[])};cache={}
    def resolve(i):
        if i not in cache:cache[i]=mul(resolve(parents[i]),local_matrix(doc['nodes'][i])) if i in parents else local_matrix(doc['nodes'][i])
        return cache[i]
    return [resolve(i) for i in range(len(doc['nodes']))]

def plan_source():
    world = Glb(WORLD)
    assert sha(world.raw) == EXPECTED_WORLD
    assert sha(VIEW.read_bytes()) == EXPECTED_VIEW
    assert sha(PLAYER_WORLD.read_bytes()) == EXPECTED_PLAYER_WORLD
    assert sha(KIT_MASTER.read_bytes()) == EXPECTED_MASTER
    assert len(world.doc['meshes']) == 1
    assert len(world.doc['meshes'][0]['primitives']) == 1
    assert len(world.doc['skins']) == 1 and len(world.doc['animations']) == 7
    p = world.doc['meshes'][0]['primitives'][0]
    assert not p.get('targets')
    attrs = {key: world.values(index) for key,index in p['attributes'].items()}
    pos = attrs['POSITION']
    ts = triangles([v[0] for v in world.values(p['indices'])])
    names = [world.doc['nodes'][i].get('name','') for i in world.doc['skins'][0]['joints']]
    head = set(); boots = set(); hands = set(); weldrows = collections.defaultdict(set)
    for i,v in enumerate(pos):
        weights = {names[j]: w for j,w in zip(attrs['JOINTS_0'][i],attrs['WEIGHTS_0'][i]) if w > 1e-5}
        if v[1] >= 1.55 or weights.get('Head',0) > 0 or weights.get('neck',0) > 0: head.add(i)
        if v[1] <= .45: boots.add(i)
        if weights.get('LeftHand',0) > 0 or weights.get('RightHand',0) > 0: hands.add(i)
        weldrows[pkey(v)].add(i)
    edgefaces = collections.defaultdict(list)
    for ti,t in enumerate(ts):
        for e in tri_edges(tuple(pkey(pos[i]) for i in t)): edgefaces[e].append(ti)
    # Preserve boundary and inherited nonmanifold neighborhoods, not a blanket
    # lock of every UV seam or all bind-pose apparent contours.
    seed_keys = {key for e,ids in edgefaces.items() if len(ids) != 2 for key in e}
    seed_rows = set().union(*(weldrows[key] for key in seed_keys))
    ring_triangles = [ti for ti,t in enumerate(ts) if any(i in seed_rows for i in t)]
    ring_keys = {pkey(pos[i]) for ti in ring_triangles for i in ts[ti]}
    ring_rows = set().union(*(weldrows[key] for key in ring_keys))
    protected = head | boots | hands | ring_rows
    fixed = [t for t in ts if any(i in protected for i in t)]
    eligible = [t for t in ts if not any(i in protected for i in t)]
    records = {key: world.records(index) for key,index in p['attributes'].items()}
    row_bytes = [b''.join(records[key][i] for key in sorted(records)) for i in range(len(pos))]
    counts = {'vertices':len(pos),'triangles':len(ts),'headNeckOrY155Rows':len(head),'bootY045Rows':len(boots),
        'handContactRows':len(hands),'boundaryNonmanifoldSeedRows':len(seed_rows),'topologyOneRingTriangles':len(ring_triangles),
        'topologyOneRingRows':len(ring_rows),'protectedRows':len(protected),'fixedTriangles':len(fixed),'eligibleTriangles':len(eligible),
        'completeAttributeRowsUnique':len(set(row_bytes)),'redundantCompleteAttributeRows':len(row_bytes)-len(set(row_bytes))}
    return world,p,attrs,ts,protected,fixed,eligible,counts,{'headNeckOrY155':head,'bootsY045':boots,'hands':hands,'topologyOneRing':ring_rows}
