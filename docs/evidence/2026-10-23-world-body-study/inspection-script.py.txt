def sha(value):return hashlib.sha256(value).hexdigest()
def canonical(triangle):return min(triangle,triangle[1:]+triangle[:1],triangle[2:]+triangle[:2])
def facts(glb,primitive):
    names=sorted(primitive['attributes'])
    records={name:glb.records(primitive['attributes'][name]) for name in names}
    rows=[b''.join(records[name][i] for name in names) for i in range(len(records['POSITION']))]
    indices=[v[0] for v in glb.values(primitive['indices'])]
    tris=recipe.triangles(indices)
    return names,rows,tris,glb.values(primitive['attributes']['POSITION'])

def geometry_quality(glb,primitive):
    pos=glb.values(primitive['attributes']['POSITION']);normals=glb.values(primitive['attributes']['NORMAL']);uv=glb.values(primitive['attributes']['TEXCOORD_0'])
    tris=recipe.triangles([v[0] for v in glb.values(primitive['indices'])])
    zero_area=zero_uv=opposite=0;minimum_area=math.inf
    for tri in tris:
        a,b,c=[pos[i] for i in tri];e=[b[k]-a[k] for k in range(3)];f=[c[k]-a[k] for k in range(3)]
        cross=[e[1]*f[2]-e[2]*f[1],e[2]*f[0]-e[0]*f[2],e[0]*f[1]-e[1]*f[0]]
        area=math.sqrt(sum(v*v for v in cross))*.5;minimum_area=min(minimum_area,area)
        zero_area+=area<=1e-14
        dot=sum(cross[k]*sum(normals[i][k] for i in tri) for k in range(3))
        opposite+=dot<0
        ua,ub,uc=[uv[i] for i in tri]
        determinant=(ub[0]-ua[0])*(uc[1]-ua[1])-(ub[1]-ua[1])*(uc[0]-ua[0])
        zero_uv+=abs(determinant)<=1e-14
    return {'triangles':len(tris),'zeroAreaAt1eMinus14SquareMetres':zero_area,'oppositeAverageVertexNormal':opposite,'zeroUvAreaAt1eMinus14':zero_uv,'minimumAreaSquareMetres':minimum_area}
