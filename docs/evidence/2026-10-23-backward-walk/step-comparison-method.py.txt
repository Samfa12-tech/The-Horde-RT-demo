EXE=ROOT/'build/presets/windows-x64-debug/Debug'/f'{TARGET}.exe'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def native_run(label):
    started=datetime.datetime.now(datetime.timezone.utc).isoformat()
    sources={s:sha(ROOT/s) for s in SOURCE}
    with (OUT/f'{label}.txt').open('wb') as log:
        then=time.monotonic()
        proc=subprocess.Popen([str(EXE)],cwd=str(EXE.parent),stdout=log,stderr=subprocess.STDOUT,creationflags=subprocess.CREATE_NO_WINDOW)
        timeout=False
        try:proc.wait(timeout=120)
        except subprocess.TimeoutExpired:timeout=True;proc.kill();proc.wait()
        wall=time.monotonic()-then
        create=ctypes.c_ulonglong();ended=ctypes.c_ulonglong();kernel=ctypes.c_ulonglong();user=ctypes.c_ulonglong()
        gettimes=ctypes.windll.kernel32.GetProcessTimes
        gettimes.argtypes=[ctypes.c_void_p,*([ctypes.POINTER(ctypes.c_ulonglong)]*4)]
        ok=gettimes(int(proc._handle),ctypes.byref(create),ctypes.byref(ended),ctypes.byref(kernel),ctypes.byref(user))
        receipt={'label':label,'startedUtc':started,'diagnosticCeilingSeconds':120,'productionCTestDeadlineSeconds':30,'gateChanged':False,'wallSeconds':wall,'kernelSeconds':kernel.value/1e7 if ok else None,'userSeconds':user.value/1e7 if ok else None,'pid':proc.pid,'exitCode':proc.returncode,'timedOut':timeout,'executableSha256':sha(EXE),'sources':sources,'scope':'direct completion diagnostic, not a CTest timeout pass'}
    (OUT/f'{label}.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(receipt),flush=True)
    return receipt

    if r.returncode:raise RuntimeError(f'{label} failed: {r.returncode}')
