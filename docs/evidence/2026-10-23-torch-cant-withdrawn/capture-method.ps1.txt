param([Parameter(Mandatory=$true)][ValidateSet('control','candidate','corrected')][string]$Version)
$ErrorActionPreference='Stop'
$root=$PWD.Path
$output=Join-Path $root "reports/torch-clearance-refinement-20261022/$Version"
if(Test-Path -LiteralPath $output){throw 'Evidence directory already exists; do not overwrite it.'}
$exe=Join-Path $root 'build/presets/windows-x64-debug/Debug/HordeLanternRT.exe'
$start=[Diagnostics.ProcessStartInfo]::new()
$start.FileName=$exe;$start.WorkingDirectory=$root;$start.UseShellExecute=$false
$start.CreateNoWindow=$true;$start.WindowStyle=[Diagnostics.ProcessWindowStyle]::Hidden
$start.RedirectStandardOutput=$true;$start.RedirectStandardError=$true
$start.Environment['HORDE_LOCAL_DARK_PROBE']='1'
foreach($argument in @('--development-rescue-journey','--capture-showcase',$output,
    '--development-checkpoint','chapter-stone-wall-corner','--capture-hidden')){$start.ArgumentList.Add($argument)}
$sourceHashes=@{}
foreach($source in @('src/gameplay/items/HeldItemKinematics.cpp','src/gameplay/HorizontalEnvelope.h',
    'src/platform/windows/DiagnosticWindow.cpp','reports/torch-clearance-refinement-20261022/route-probe.inl')){
    $sourceHashes[$source]=(Get-FileHash -Algorithm SHA256 -LiteralPath (Join-Path $root $source)).Hash
}
$receipt=@{executableSha256=(Get-FileHash -Algorithm SHA256 -LiteralPath $exe).Hash;sources=$sourceHashes;
    startedUtc=[DateTime]::UtcNow.ToString('o');arguments=@($start.ArgumentList);version=$Version;
    scope='Debug input-driven current chapter, frozen look images; not natural journey, physical controls or phone acceptance'}
$process=[Diagnostics.Process]::Start($start)
$out=$process.StandardOutput.ReadToEndAsync();$err=$process.StandardError.ReadToEndAsync()
$timer=[Diagnostics.Stopwatch]::StartNew();$timedOut=$false
while(-not $process.WaitForExit(1000)){
    if($timer.Elapsed.TotalSeconds -gt 180){$timedOut=$true;$process.Kill();$process.WaitForExit();break}
}
$receipt.pid=$process.Id;$receipt.exitCode=$process.ExitCode;$receipt.timedOut=$timedOut
$receipt.elapsedSeconds=$timer.Elapsed.TotalSeconds;$receipt.finishedUtc=[DateTime]::UtcNow.ToString('o')
[IO.File]::WriteAllText($output+'.stdout.txt',$out.Result)
[IO.File]::WriteAllText($output+'.stderr.txt',$err.Result)
$receipt | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath ($output+'.process.json') -Encoding utf8
Write-Output "$Version capture exit=$($process.ExitCode), timeout=$timedOut, elapsed=$($timer.Elapsed.TotalSeconds)s"
$result=$process.ExitCode;$process.Dispose()
if($timedOut){throw 'Capture exceeded unchanged180s local diagnostic ceiling.'}
exit $result
