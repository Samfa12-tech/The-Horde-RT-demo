cmake_minimum_required(VERSION 3.22)
project(HordeHybridLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(HORDE_SOURCE "" CACHE PATH "Existing Horde checkout (read only)")
if(NOT EXISTS "${HORDE_SOURCE}/src/scene/TombStoneBevels.h")
  message(FATAL_ERROR "Set HORDE_SOURCE to the existing Horde checkout; no clone is needed.")
endif()
if(ANDROID)
  add_library(horde_hybrid_lab SHARED Lab.cpp Android.cpp)
  target_link_libraries(horde_hybrid_lab PRIVATE vulkan log)
else()
  find_package(Vulkan REQUIRED)
  add_executable(horde_hybrid_lab Lab.cpp Main.cpp)
  target_link_libraries(horde_hybrid_lab PRIVATE Vulkan::Vulkan)
  if(MSVC)
    target_compile_options(horde_hybrid_lab PRIVATE /W4 /EHsc)
  endif()
endif()
target_include_directories(horde_hybrid_lab PRIVATE "${HORDE_SOURCE}/src" "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(identity "")
foreach(path IN ITEMS Lab.cpp Lab.h Main.cpp Android.cpp generated/Shaders.h CMakeLists.txt)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" digest)
  string(APPEND identity "${path}:${digest}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
foreach(path IN ITEMS src/scene/TombStoneBevels.h src/scene/ShowcaseOverheadGeometry.h src/gameplay/ShowcaseRoute.h src/vulkan/raytracing/RtSceneAbi.generated.h)
  file(SHA256 "${HORDE_SOURCE}/${path}" digest)
  string(APPEND identity "${path}:${digest}\n")
endforeach()
string(SHA256 build_identity "${identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/BuildIdentity.h" "#pragma once\ninline constexpr char kLabBuildIdentity[]=\"${build_identity}\";\n")
target_include_directories(horde_hybrid_lab PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
